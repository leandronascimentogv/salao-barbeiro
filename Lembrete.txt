/**
 * Lembrete.gs — Confirmação e lembrete por e-mail (MailApp)
 */

const ASSUNTO_CONFIRMACAO = 'Agendamento confirmado';
const ASSUNTO_LEMBRETE = 'Lembrete do seu horário';

function emailDoCliente(telefone) {
  const c = lerAba(ABAS.CLIENTES).find(x => String(x.telefone) === String(telefone));
  return c ? String(c.email || '') : '';
}

function nomeServico(id) {
  const s = lerAba(ABAS.SERVICOS).find(x => String(x.id) === String(id));
  return s ? s.nome : 'Serviço';
}

function corpoMensagem(a, titulo) {
  const cfg = getConfig();
  const data = String(a.data).slice(0, 10).split('-').reverse().join('/');
  return [
    titulo,
    '',
    'Olá, ' + a.cliente_nome + '!',
    '',
    'Seu horário na ' + (cfg.nome_estabelecimento || 'nossa loja') + ':',
    'Serviço: ' + nomeServico(a.servico_id),
    'Data: ' + data,
    'Hora: ' + String(a.hora_inicio).slice(0, 5),
    '',
    'Endereço: ' + (cfg.endereco || ''),
    'Telefone: ' + (cfg.telefone || ''),
    '',
    'Se precisar remarcar, responda este e-mail.'
  ].join('\n');
}

function enviarConfirmacao(agendamentoId) {
  const a = lerAba(ABAS.AGENDAMENTOS).find(x => String(x.id) === String(agendamentoId));
  if (!a) return 'Agendamento não encontrado';
  const email = emailDoCliente(a.cliente_telefone);
  if (!email) return 'Cliente sem e-mail cadastrado';
  MailApp.sendEmail(email, ASSUNTO_CONFIRMACAO, corpoMensagem(a, 'Agendamento confirmado'));
  return 'Confirmação enviada para ' + email;
}

function enviarLembrete(agendamentoId) {
  const a = lerAba(ABAS.AGENDAMENTOS).find(x => String(x.id) === String(agendamentoId));
  if (!a) return false;
  const email = emailDoCliente(a.cliente_telefone);
  if (!email) return false;
  MailApp.sendEmail(email, ASSUNTO_LEMBRETE, corpoMensagem(a, 'Lembrete do seu horário'));
  return true;
}

function enviarLembretesAmanha() {
  const amanha = new Date();
  amanha.setDate(amanha.getDate() + 1);
  const iso = Utilities.formatDate(amanha, Session.getScriptTimeZone(), 'yyyy-MM-dd');
  const lista = lerAba(ABAS.AGENDAMENTOS).filter(a =>
    String(a.data).slice(0, 10) === iso &&
    String(a.status).toLowerCase() === STATUS.AGENDADO);
  let enviados = 0;
  lista.forEach(a => { if (enviarLembrete(a.id)) enviados++; });
  return 'Lembretes enviados: ' + enviados;
}

function criarGatilhoLembretes() {
  ScriptApp.getProjectTriggers().forEach(t => {
    if (t.getHandlerFunction() === 'enviarLembretesAmanha') ScriptApp.deleteTrigger(t);
  });
  ScriptApp.newTrigger('enviarLembretesAmanha')
    .timeBased().atHour(18).everyDays(1).create();
  return 'Gatilho diário criado para as 18h';
}
