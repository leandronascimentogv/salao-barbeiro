/**
 * Dados.gs — Leitura e escrita nas abas
 */

function lerAba(nome) {
  const sh = SpreadsheetApp.getActive().getSheetByName(nome);
  if (!sh) throw new Error('Aba não encontrada: ' + nome);
  const valores = sh.getDataRange().getValues();
  if (valores.length < 2) return [];
  const cab = valores.shift().map(String);
  return valores.map(l => {
    const o = {};
    cab.forEach((c, i) => o[c] = l[i]);
    return o;
  });
}

function proximoId(nome) {
  const itens = lerAba(nome);
  return itens.length ? Math.max(...itens.map(i => Number(i.id) || 0)) + 1 : 1;
}

function appendLinha(nome, obj) {
  const sh = SpreadsheetApp.getActive().getSheetByName(nome);
  const cab = sh.getRange(1, 1, 1, sh.getLastColumn()).getValues()[0].map(String);
  sh.appendRow(cab.map(c => obj[c] !== undefined ? obj[c] : ''));
  return obj;
}

function atualizarCelula(nome, id, coluna, valor) {
  const sh = SpreadsheetApp.getActive().getSheetByName(nome);
  const dados = sh.getDataRange().getValues();
  const cab = dados[0].map(String);
  const cId = cab.indexOf('id') + 1;
  const cCol = cab.indexOf(coluna) + 1;
  if (cId === 0 || cCol === 0) return false;
  for (let i = 1; i < dados.length; i++) {
    if (String(dados[i][cId - 1]) === String(id)) {
      sh.getRange(i + 1, cCol).setValue(valor);
      return true;
    }
  }
  return false;
}
