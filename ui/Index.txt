<script>
function acao(nome){
  document.getElementById('saida').textContent='Executando...';
  google.script.run.withSuccessHandler(r=>{
    document.getElementById('saida').textContent = typeof r==='string' ? r : JSON.stringify(r,null,2);
  }).executarAcao(nome);
}
function carregarAgenda(){
  const d=document.getElementById('data').value;
  if(!d) return;
  document.getElementById('agenda').textContent='Carregando...';
  google.script.run.withSuccessHandler(lista=>{
    if(!lista.length){document.getElementById('agenda').innerHTML='<em>Nenhum agendamento nesse dia.</em>';return;}
    document.getElementById('agenda').innerHTML = lista.map(a=>`
      <div class="item">
        <div><span class="hora">${a.hora_inicio}</span> — ${a.cliente_nome}</div>
        <div><span class="st ${a.status}">${a.status}</span> · R$ ${a.valor}</div>
        <div>
          <button class="mini" onclick="mudar('${a.id}','concluido')">Concluir</button>
          <button class="mini sec" onclick="mudar('${a.id}','cancelado')">Cancelar</button>
          <button class="mini sec" onclick="mudar('${a.id}','faltou')">Faltou</button>
          <button class="mini sec" onclick="remarcar('${a.id}')">Remarcar</button>
        </div>
      </div>`).join('');
  }).apiListarAgendamentos(d);
}
function mudar(id,status){
  google.script.run.withSuccessHandler(()=>carregarAgenda()).apiAtualizarStatus(id,status);
}
function remarcar(id){
  const novaData=prompt('Nova data (AAAA-MM-DD):');
  if(!novaData) return;
  const novaHora=prompt('Nova hora (HH:MM):');
  if(!novaHora) return;
  google.script.run.withSuccessHandler(r=>{
    if(!r.ok) alert(r.erro);
    carregarAgenda();
  }).apiRemarcar(id,novaData,novaHora);
}
google.script.run.withSuccessHandler(t=>document.getElementById('titulo').textContent='Painel — '+t).getNomeEstabelecimento();
const d=new Date();
document.getElementById('data').value = d.getFullYear()+'-'+String(d.getMonth()+1).padStart(2,'0')+'-'+String(d.getDate()).padStart(2,'0');
carregarAgenda();
