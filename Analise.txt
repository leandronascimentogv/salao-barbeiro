/**
 * Analise.gs — Motor de análise
 */

function paraData(v) {
  return v instanceof Date ? v : new Date(v);
}

function calcularMetricas(inicio, fim) {
  const ags = lerAba(ABAS.AGENDAMENTOS).filter(a => {
    const d = paraData(a.data);
    return d >= inicio && d <= fim;
  });
  const servicos = {};
  lerAba(ABAS.SERVICOS).forEach(s => servicos[s.id] = s);

  let receita = 0, concluidos = 0, cancelados = 0, faltas = 0;
  const porServico = {}, porProfissional = {}, porDia = {}, porHora = {};
  const clientesSet = new Set();

  ags.forEach(a => {
    const status = String(a.status || '').toLowerCase();
    const valor = Number(a.valor) || 0;
    if (status === 'concluido' || status === 'concluído') {
      receita += valor;
      concluidos++;
      const s = servicos[a.servico_id] ? servicos[a.servico_id].nome : a.servico_id;
      porServico[s] = (porServico[s] || 0) + valor;
      porProfissional[a.profissional_id] = (porProfissional[a.profissional_id] || 0) + valor;
      const dia = Utilities.formatDate(paraData(a.data), Session.getScriptTimeZone(), 'yyyy-MM-dd');
      porDia[dia] = (porDia[dia] || 0) + valor;
      const hora = String(a.hora_inicio).slice(0, 5);
      porHora[hora] = (porHora[hora] || 0) + 1;
      clientesSet.add(a.cliente_nome);
    } else if (status === 'cancelado') {
      cancelados++;
    } else if (status === 'faltou') {
      faltas++;
    }
  });

  const total = ags.length || 1;
  return {
    periodo: {
      inicio: Utilities.formatDate(inicio, Session.getScriptTimeZone(), 'yyyy-MM-dd'),
      fim: Utilities.formatDate(fim, Session.getScriptTimeZone(), 'yyyy-MM-dd')
    },
    receita_total: receita,
    ticket_medio: concluidos ? receita / concluidos : 0,
    agendamentos_total: ags.length,
    concluidos: concluidos,
    cancelados: cancelados,
    faltas: faltas,
    taxa_no_show: faltas / total,
    taxa_cancelamento: cancelados / total,
    clientes_unicos: clientesSet.size,
    receita_por_servico: porServico,
    receita_por_profissional: porProfissional,
    receita_por_dia: porDia,
    horarios_de_pico: porHora
  };
}

function gerarAnalises() {
  const hoje = new Date();
  const inicio = new Date(hoje.getFullYear(), hoje.getMonth(), 1);
  const fim = new Date(hoje.getFullYear(), hoje.getMonth() + 1, 0, 23, 59, 59);
  const m = calcularMetricas(inicio, fim);

  const ss = SpreadsheetApp.getActive();
  let sh = ss.getSheetByName(ABAS.ANALISES);
  if (!sh) sh = ss.insertSheet(ABAS.ANALISES);
  sh.clear();

  const linhas = [
    ['Metrica', 'Valor'],
    ['Periodo', m.periodo.inicio + ' a ' + m.periodo.fim],
    ['Receita total', m.receita_total],
    ['Ticket medio', m.ticket_medio],
    ['Agendamentos', m.agendamentos_total],
    ['Concluidos', m.concluidos],
    ['Cancelados', m.cancelados],
    ['Faltas', m.faltas],
    ['Taxa de no-show', m.taxa_no_show],
    ['Taxa de cancelamento', m.taxa_cancelamento],
    ['Clientes unicos', m.clientes_unicos],
    [],
    ['Receita por servico', ''],
    ...Object.entries(m.receita_por_servico),
    [],
    ['Receita por profissional', ''],
    ...Object.entries(m.receita_por_profissional),
    [],
    ['Horarios de pico', ''],
    ...Object.entries(m.horarios_de_pico)
  ];
  sh.getRange(1, 1, linhas.length, 2).setValues(linhas);
  sh.getRange(1, 1, 1, 2).setFontWeight('bold');
  return m;
}
